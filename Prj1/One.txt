    <script>
        // 配置信息
        const config = {
            token: "version=2018-10-31&res=products%2FAceV67og0i%2Fdevices%2Fmytext&et=1805693871&method=md5&sign=PEUKCtejaA%2By49Qw78sOfw%3D%3D",
            productId: "AceV67og0i",
            deviceName: "mytext",
            api: {
                deviceDetail: "https://iot-api.heclouds.com/device/detail",
                getProperty: "https://iot-api.heclouds.com/thingmodel/query-device-property",
                setProperty: "https://iot-api.heclouds.com/thingmodel/set-device-property",
                getHistory: "https://iot-api.heclouds.com/thingmodel/query-device-property-history"
            }
        };

        let chartInstance = null;

        // 获取设备在线状态
        async function getDeviceStatus() {
            try {
                const res = await fetch(`${config.api.deviceDetail}?product_id=${config.productId}&device_name=${config.deviceName}`, {
                    headers: { "authorization": config.token }
                });
                const data = await res.json();
                if (data.code === 0) {
                    const status = data.data.status;
                    const dot = document.getElementById("statusDot");
                    const text = document.getElementById("statusText");
                    if (status === 1) {
                        dot.className = "w-3 h-3 rounded-full bg-green-500";
                        text.textContent = "在线";
                        text.className = "text-green-600";
                    } else {
                        dot.className = "w-3 h-3 rounded-full bg-gray-400";
                        text.textContent = "离线";
                        text.className = "text-gray-600";
                    }
                }
            } catch (err) {
                console.error("获取设备状态失败：", err);
            }
        }

        // 获取设备当前属性
        async function getDeviceProperties() {
            try {
                const res = await fetch(`${config.api.getProperty}?product_id=${config.productId}&device_name=${config.deviceName}`, {
                    headers: { "authorization": config.token }
                });
                const data = await res.json();
                if (data.code !== 0) {
                    console.error("获取属性失败：", data.msg);
                    return;
                }
                const list = data.data;
                
                // 解析各个属性值
                const temperature = list.find(i => i.identifier === "temperature")?.value || 0;
                const humidity = list.find(i => i.identifier === "humidity")?.value || 0;
                const co2 = list.find(i => i.identifier === "co2")?.value || 0;
                const led = list.find(i => i.identifier === "LED")?.value || "-";
                const key = list.find(i => i.identifier === "key")?.value || false;

                // 更新页面显示
                document.getElementById("temperature").textContent = parseFloat(temperature).toFixed(1);
                document.getElementById("humidity").textContent = parseFloat(humidity).toFixed(1);
                document.getElementById("co2").textContent = parseInt(co2);
                document.getElementById("LED").textContent = led;
                document.getElementById("keyStatus").textContent = key ? "开启" : "关闭";
            } catch (err) {
                console.error("获取属性失败：", err);
            }
        }

        // 控制开关
        function controlSwitch(status) {
            // 直接显示已执行，无需等待响应
            document.getElementById("controlTip").textContent = "已执行";
            setTimeout(() => {
                document.getElementById("controlTip").textContent = "";
            }, 2000);

            // 后台发送控制指令
            fetch(config.api.setProperty, {
                method: "POST",
                headers: {
                    "authorization": config.token,
                    "Content-Type": "application/json"
                },
                body: JSON.stringify({
                    product_id: config.productId,
                    device_name: config.deviceName,
                    params: { "key": status }
                })
            })
            .then(res => res.json())
            .then(() => {
                // 指令发送后刷新数据
                getDeviceProperties();
            })
            .catch(err => {
                console.error("控制指令发送失败：", err);
            });
        }

        // 加载历史数据
        async function loadHistoryData(identifier, name, unit, color) {
            const modal = document.getElementById("historyModal");
            const title = document.getElementById("modalTitle");
            title.textContent = `${name}历史数据`;
            modal.classList.remove("hidden");

            // 初始化图表
            if (!chartInstance) {
                chartInstance = echarts.init(document.getElementById("historyChart"));
            }
            chartInstance.showLoading();

            try {
                // 查询最近24小时的历史数据
                const endTime = Date.now();
                const startTime = endTime - 24 * 60 * 60 * 1000;

                const res = await fetch(`${config.api.getHistory}?product_id=${config.productId}&device_name=${config.deviceName}&identifier=${identifier}&start_time=${startTime}&end_time=${endTime}&limit=100`, {
                    headers: { "authorization": config.token }
                });
                const data = await res.json();
                if (data.code !== 0) {
                    alert("获取历史数据失败：" + data.msg);
                    chartInstance.hideLoading();
                    return;
                }

                const list = data.data.list || [];
                // 反转数据，按时间正序排列
                list.reverse();

                // 处理数据格式
                const times = list.map(item => {
                    const date = new Date(parseInt(item.time));
                    return `${date.getHours().toString().padStart(2, '0')}:${date.getMinutes().toString().padStart(2, '0')}`;
                });
                const values = list.map(item => {
                    const val = item.value;
                    return isNaN(val) ? val : parseFloat(val);
                });

                // 渲染图表
                const option = {
                    tooltip: {
                        trigger: 'axis'
                    },
                    grid: {
                        left: '3%',
                        right: '4%',
                        bottom: '3%',
                        containLabel: true
                    },
                    xAxis: {
                        type: 'category',
                        data: times,
                        axisLabel: {
                            rotate: 30
                        }
                    },
                    yAxis: {
                        type: 'value',
                        name: unit
                    },
                    series: [
                        {
                            name: name,
                            type: 'line',
                            data: values,
                            smooth: true,
                            itemStyle: {
                                color: color
                            },
                            areaStyle: {
                                color: new echarts.graphic.LinearGradient(0, 0, 0, 1, [
                                    { offset: 0, color: color + '80' },
                                    { offset: 1, color: color + '10' }
                                ])
                            }
                        }
                    ]
                };
                chartInstance.setOption(option, true);
                chartInstance.hideLoading();

                // 窗口大小变化时自适应图表
                window.addEventListener('resize', () => {
                    chartInstance.resize();
                });
            } catch (err) {
                console.error("获取历史数据失败：", err);
                alert("获取历史数据失败，请稍后重试");
                chartInstance.hideLoading();
            }
        }

        // 页面初始化
        window.onload = () => {
            // 初始加载数据
            getDeviceStatus();
            getDeviceProperties();

            // 定时刷新数据，每10秒一次
            setInterval(() => {
                getDeviceStatus();
                getDeviceProperties();
            }, 10000);

            // 绑定控制按钮事件
            document.getElementById("openBtn").addEventListener("click", () => {
                controlSwitch(true);
            });
            document.getElementById("closeBtn").addEventListener("click", () => {
                controlSwitch(false);
            });

            // 绑定属性卡片点击事件
            document.querySelectorAll(".property-card").forEach(card => {
                card.addEventListener("click", () => {
                    const identifier = card.dataset.identifier;
                    const name = card.dataset.name;
                    const unit = card.dataset.unit;
                    const color = card.dataset.color;
                    loadHistoryData(identifier, name, unit, color);
                });
            });

            // 绑定模态框关闭事件
            document.getElementById("closeModal").addEventListener("click", () => {
                document.getElementById("historyModal").classList.add("hidden");
            });
            document.getElementById("historyModal").addEventListener("click", (e) => {
                if (e.target === document.getElementById("historyModal")) {
                    document.getElementById("historyModal").classList.add("hidden");
                }
            });
        };
    </script>
